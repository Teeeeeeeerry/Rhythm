cmake_minimum_required(VERSION 3.21)

# Version comes from the workspace manifest, not from a copy kept here (#255).
# Cargo.toml's [workspace.package] version is the single source; this file used
# to hold a hand-maintained duplicate that had drifted 39 patch releases behind.
# Configure fails loudly if the version cannot be read - a silent fallback would
# ship an app whose reported version is wrong, which is the bug being fixed.
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/../Cargo.toml" RHYTHM_WORKSPACE_MANIFEST)
string(REGEX MATCH
    "\\[workspace\\.package\\][^[]*version[ \t]*=[ \t]*\"([0-9]+\\.[0-9]+\\.[0-9]+)\""
    RHYTHM_VERSION_MATCH "${RHYTHM_WORKSPACE_MANIFEST}")
if(NOT CMAKE_MATCH_1)
    message(FATAL_ERROR
        "cannot read [workspace.package] version from ../Cargo.toml")
endif()
set(RHYTHM_VERSION "${CMAKE_MATCH_1}")

project(Rhythm VERSION ${RHYTHM_VERSION} LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

# Windows SDK
set(CMAKE_VS_WINDOWS_TARGET_PLATFORM_VERSION "10.0")

# The behaviour library -- everything that is not XAML shell -- is declared in
# one module (#330): its sources, usage requirements and the Rust core. The
# module pulls in the pinned dependencies (RhythmWindowsDeps.cmake, #386).
include(${CMAKE_CURRENT_SOURCE_DIR}/cmake/RhythmBehavior.cmake)

# What the MSBuild app project needs from this build (#428): the pinned
# packages plus the behaviour library's file, include path, defines, options
# and the core -- every value derived from the targets above, never spelled
# out again (#330). One file per configuration; Rhythm/Rhythm.vcxproj imports it.
string(CONFIGURE [=[
<?xml version="1.0" encoding="utf-8"?>
<!-- Generated by windows/CMakeLists.txt - do not edit. -->
<Project xmlns="http://schemas.microsoft.com/developer/msbuild/2003">
  <Import Project="@RHYTHM_WINDOWS_DEPS_DIR@/RhythmWindowsDeps.props" />
  <PropertyGroup>
    <RhythmBehaviorLib>$<TARGET_FILE:RhythmBehavior></RhythmBehaviorLib>
    <RhythmCoreImportLib>$<TARGET_LINKER_FILE:Rhythm::Core></RhythmCoreImportLib>
    <RhythmCoreDll>$<TARGET_FILE:Rhythm::Core></RhythmCoreDll>
    <RhythmBehaviorIncludes>$<TARGET_PROPERTY:RhythmBehavior,INTERFACE_INCLUDE_DIRECTORIES></RhythmBehaviorIncludes>
    <RhythmBehaviorDefines>$<TARGET_PROPERTY:RhythmBehavior,INTERFACE_COMPILE_DEFINITIONS></RhythmBehaviorDefines>
    <RhythmBehaviorOptions>$<JOIN:$<TARGET_PROPERTY:RhythmBehavior,INTERFACE_COMPILE_OPTIONS>, ></RhythmBehaviorOptions>
  </PropertyGroup>
</Project>
]=] RHYTHM_BEHAVIOR_PROPS @ONLY)
file(GENERATE OUTPUT "${CMAKE_BINARY_DIR}/$<CONFIG>/RhythmBehavior.props"
    CONTENT "${RHYTHM_BEHAVIOR_PROPS}")

# ─── Tests (Wave 4a) ────────────────────────────────────────────────

# Catch2 v3 header-only test target, linking the behaviour library (#428).
enable_testing()

set(TEST_SOURCES
    tests/TestHelpers.h
    # #413: GeneratedCodec.h included alone must compile.
    tests/GeneratedCodecStandalone.cpp
    # #329: behaviour headers included alone pull in no WinUI projection.
    tests/BehaviorHeadersStandalone.cpp
    tests/vendor/catch_amalgamated.cpp
    tests/AppStateBehaviorTests.cpp
    tests/BridgeBehaviorTests.cpp
    tests/L10nTests.cpp
    tests/ViewStateTests.cpp
    # #455: the fixtures' own contract (unique, cleanable temp directories).
    tests/TestFixtureTests.cpp
    # #365: every contract object round-trips through the generated codec.
    tests/ContractRoundTripTests.cpp
)

# Catch2's own main() (catch_amalgamated.cpp): AppState is plain C++ (#326),
# so no WinRT apartment has to be initialised before the suites run.
add_executable(RhythmTests ${TEST_SOURCES})

target_include_directories(RhythmTests PRIVATE
    ${CMAKE_CURRENT_SOURCE_DIR}/tests/vendor
)

target_link_libraries(RhythmTests PRIVATE RhythmBehavior)

# #365: the round-trip suite reads the contract itself to learn which objects
# exist, so an object added there is covered without editing the test.
target_compile_definitions(RhythmTests PRIVATE
    RHYTHM_FFI_CONTRACT="${CMAKE_CURRENT_SOURCE_DIR}/../contracts/ffi-contract.json")

# Runtime dependencies (the core DLL) next to the test binary (#330).
rhythm_copy_runtime_dlls(RhythmTests)

add_test(NAME RhythmTests COMMAND RhythmTests)
