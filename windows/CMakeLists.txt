cmake_minimum_required(VERSION 3.20)

# Version comes from the workspace manifest, not from a copy kept here (#255).
# Cargo.toml's [workspace.package] version is the single source; this file used
# to hold a hand-maintained duplicate that had drifted 39 patch releases behind.
# Configure fails loudly if the version cannot be read - a silent fallback would
# ship an app whose reported version is wrong, which is the bug being fixed.
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/../Cargo.toml" RHYTHM_WORKSPACE_MANIFEST)
string(REGEX MATCH
    "\\[workspace\\.package\\][^[]*version[ \t]*=[ \t]*\"([0-9]+\\.[0-9]+\\.[0-9]+)\""
    RHYTHM_VERSION_MATCH "${RHYTHM_WORKSPACE_MANIFEST}")
if(NOT CMAKE_MATCH_1)
    message(FATAL_ERROR
        "cannot read [workspace.package] version from ../Cargo.toml")
endif()
set(RHYTHM_VERSION "${CMAKE_MATCH_1}")

project(Rhythm VERSION ${RHYTHM_VERSION} LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

# Windows SDK
set(CMAKE_VS_WINDOWS_TARGET_PLATFORM_VERSION "10.0")

# Windows App SDK (C++/WinRT projection + import libs) and nlohmann_json are
# fetched at pinned versions and exposed as imported targets by one module
# (#386): upstream ships no CMake package for the SDK, so find_package could
# never succeed on a clean machine. The same module writes the props file the
# MSBuild app project imports, so both builds share one set of pins (#428).
include(${CMAKE_CURRENT_SOURCE_DIR}/cmake/RhythmWindowsDeps.cmake)

# ─── Behaviour library (ADR-0004, #428) ─────────────────────────────

# Everything that is not XAML shell: app state, the core bridge and the L10n
# layer. The app (MSBuild, Rhythm/Rhythm.vcxproj) and the test host both link
# it -- these sources are declared here and nowhere else.
set(BEHAVIOR_SOURCES
    Rhythm/pch.h
    Rhythm/L10n.h
    Rhythm/L10nAccessors.h
    Rhythm/AppState.h
    Rhythm/AppState.cpp
    Rhythm/Bridge/GeneratedCodec.h
    Rhythm/Bridge/L10nKeys.h
    Rhythm/Bridge/MessageSpec.h
    Rhythm/Bridge/RhythmCore.h
    Rhythm/Bridge/RhythmCore.cpp
    Rhythm/Bridge/rhythm_core.h
)

add_library(RhythmBehavior STATIC ${BEHAVIOR_SOURCES})

# Bridge/ holds the core's C header, included as <rhythm_core.h> (#408).
target_include_directories(RhythmBehavior PUBLIC
    ${CMAKE_CURRENT_SOURCE_DIR}/Rhythm
    ${CMAKE_CURRENT_SOURCE_DIR}/Rhythm/Bridge
)

# <Windows.h> defines min/max macros that break std::min/max and Catch2 (#408).
target_compile_definitions(RhythmBehavior PUBLIC NOMINMAX)

# The L10n layer holds the remaining Chinese literals (localized fallback);
# force UTF-8 so MSVC reads them regardless of the system code page.
if(MSVC)
    target_compile_options(RhythmBehavior PUBLIC /utf-8)
endif()

# L10n.h includes Bridge/MessageSpec.h, which parses the core's message spec
# JSON; the core itself is a DLL behind its import library.
target_link_libraries(RhythmBehavior PUBLIC
    Microsoft.WindowsAppSDK
    nlohmann_json::nlohmann_json
    ${CMAKE_CURRENT_SOURCE_DIR}/../target/release/rhythm_core.dll.lib
)


# What the MSBuild app project needs from this build: the pinned packages and
# the behaviour library's include path, defines and library (#428). One file,
# written per configuration; Rhythm/Rhythm.vcxproj imports it.
set(RHYTHM_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}")
get_filename_component(RHYTHM_CORE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../target/release" ABSOLUTE)
string(CONFIGURE [=[
<?xml version="1.0" encoding="utf-8"?>
<!-- Generated by windows/CMakeLists.txt - do not edit. -->
<Project xmlns="http://schemas.microsoft.com/developer/msbuild/2003">
  <Import Project="@RHYTHM_WINDOWS_DEPS_DIR@/RhythmWindowsDeps.props" />
  <PropertyGroup>
    <RhythmBehaviorLib>$<TARGET_FILE:RhythmBehavior></RhythmBehaviorLib>
    <RhythmCoreImportLib>@RHYTHM_CORE_DIR@/rhythm_core.dll.lib</RhythmCoreImportLib>
    <RhythmCoreDll>@RHYTHM_CORE_DIR@/rhythm_core.dll</RhythmCoreDll>
    <RhythmBehaviorIncludes>@RHYTHM_SOURCE_DIR@/Rhythm;@RHYTHM_SOURCE_DIR@/Rhythm/Bridge;$(RhythmJsonInclude)</RhythmBehaviorIncludes>
  </PropertyGroup>
</Project>
]=] RHYTHM_BEHAVIOR_PROPS @ONLY)
file(GENERATE OUTPUT "${CMAKE_BINARY_DIR}/$<CONFIG>/RhythmBehavior.props"
    CONTENT "${RHYTHM_BEHAVIOR_PROPS}")

# ─── Tests (Wave 4a) ────────────────────────────────────────────────

# Catch2 v3 header-only test target, linking the behaviour library (#428).
enable_testing()

set(TEST_SOURCES
    tests/TestMain.cpp
    tests/TestHelpers.h
    # #413: GeneratedCodec.h included alone must compile.
    tests/GeneratedCodecStandalone.cpp
    tests/vendor/catch_amalgamated.cpp
    tests/AppStateBehaviorTests.cpp
    tests/BridgeBehaviorTests.cpp
    tests/L10nTests.cpp
)

add_executable(RhythmTests ${TEST_SOURCES})

# catch_amalgamated.cpp ships its own main(); TestMain.cpp provides ours
# (with init_apartment), so tell Catch2 to stand down.
target_compile_definitions(RhythmTests PRIVATE CATCH_AMALGAMATED_CUSTOM_MAIN)

target_include_directories(RhythmTests PRIVATE
    ${CMAKE_CURRENT_SOURCE_DIR}/tests/vendor
)

target_link_libraries(RhythmTests PRIVATE RhythmBehavior)

# Runtime dependencies for the test binary.
add_custom_command(TARGET RhythmTests POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy_if_different
        ${CMAKE_CURRENT_SOURCE_DIR}/../target/release/rhythm_core.dll
        $<TARGET_FILE_DIR:RhythmTests>
)

add_test(NAME RhythmTests COMMAND RhythmTests)
